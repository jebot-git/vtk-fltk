cmake_minimum_required(VERSION 3.24)
project(vtk_fltk LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
# System FLTK shared libraries; no Qt or GTK frontend is requested.
find_path(FLTK_INCLUDE_DIR FL/Fl.H REQUIRED)
find_library(FLTK_LIBRARY fltk REQUIRED)
find_library(FLTK_GL_LIBRARY fltk_gl REQUIRED)
find_path(VTK_INCLUDE_DIR vtkGenericOpenGLRenderWindow.h PATH_SUFFIXES vtk vtk-9.5 REQUIRED)
find_package(OpenGL REQUIRED)
add_executable(vtk-fltk main.cpp)
target_include_directories(vtk-fltk PRIVATE ${FLTK_INCLUDE_DIR} ${VTK_INCLUDE_DIR})
target_link_libraries(vtk-fltk PRIVATE ${FLTK_GL_LIBRARY} ${FLTK_LIBRARY} OpenGL::GL)
foreach(module CommonCore CommonDataModel CommonExecutionModel FiltersSources RenderingCore RenderingOpenGL2 InteractionStyle RenderingUI)
 find_library(VTK_${module} NAMES vtk${module} libvtk${module}.so.1 REQUIRED)
 target_link_libraries(vtk-fltk PRIVATE ${VTK_${module}})
endforeach()
find_library(VTK_SYS NAMES vtksys libvtksys.so.1 REQUIRED)
target_link_libraries(vtk-fltk PRIVATE ${VTK_SYS})
