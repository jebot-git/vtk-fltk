cmake_minimum_required(VERSION 3.24)
project(vtk_fltk VERSION 0.2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
include(GNUInstallDirs)
find_path(FLTK_INCLUDE_DIR FL/Fl.H REQUIRED)
find_library(FLTK_LIBRARY fltk REQUIRED)
find_library(FLTK_GL_LIBRARY fltk_gl REQUIRED)
find_library(FLTK_IMAGES_LIBRARY fltk_images REQUIRED)
find_path(VTK_INCLUDE_DIR vtkGenericOpenGLRenderWindow.h PATH_SUFFIXES vtk vtk-9.5 REQUIRED)
find_package(OpenGL REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(FONTCONFIG REQUIRED IMPORTED_TARGET fontconfig zlib)
set(VTK_LIBRARIES)
foreach(module CommonCore CommonDataModel CommonExecutionModel FiltersSources RenderingCore RenderingOpenGL2 InteractionStyle RenderingUI IOLegacy IOXML IOXMLParser IOGeometry IOPLY)
 find_library(VTK_${module} NAMES vtk${module} libvtk${module}.so.1 REQUIRED)
 list(APPEND VTK_LIBRARIES "${VTK_${module}}")
endforeach()
find_library(VTK_SYS NAMES vtksys libvtksys.so.1 REQUIRED)
list(APPEND VTK_LIBRARIES "${VTK_SYS}")
add_library(vtk-fltk-adapter SHARED src/RenderWidget.cpp)
set_target_properties(vtk-fltk-adapter PROPERTIES OUTPUT_NAME vtk-fltk SOVERSION 0 VERSION ${PROJECT_VERSION})
target_include_directories(vtk-fltk-adapter PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}> PRIVATE "${VTK_INCLUDE_DIR}" "${FLTK_INCLUDE_DIR}")
target_link_libraries(vtk-fltk-adapter PUBLIC ${FLTK_GL_LIBRARY} ${FLTK_LIBRARY} PRIVATE OpenGL::GL ${VTK_LIBRARIES})
add_executable(vtk-fltk main.cpp)
target_include_directories(vtk-fltk PRIVATE ${FLTK_INCLUDE_DIR} ${VTK_INCLUDE_DIR})
target_link_libraries(vtk-fltk PRIVATE vtk-fltk-adapter ${FLTK_GL_LIBRARY} ${FLTK_LIBRARY} ${FLTK_IMAGES_LIBRARY} OpenGL::GL ${VTK_LIBRARIES} PkgConfig::FONTCONFIG ${CMAKE_DL_LIBS})
include(suite/desktop-menu.cmake)
de_xr_desktop_menu(vtk-fltk)
install(TARGETS vtk-fltk-adapter EXPORT VTKFLTKTargets LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(TARGETS vtk-fltk RUNTIME DESTINATION ${CMAKE_INSTALL_LIBEXECDIR}/vtk-fltk)
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(EXPORT VTKFLTKTargets NAMESPACE VTKFLTK:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/VTKFLTK)
install(FILES cmake/VTKFLTKConfig.cmake DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/VTKFLTK)
install(FILES README.md LICENSE DESTINATION ${CMAKE_INSTALL_DATADIR}/doc/vtk-fltk)
